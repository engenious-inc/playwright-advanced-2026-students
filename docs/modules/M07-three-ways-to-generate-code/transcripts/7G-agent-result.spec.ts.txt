import { test, expect } from '../../shared/fixtures/index.js';
import { skipInHeadless, skipOnMobile } from '../../shared/test-guards.js';

test(
  'Browse menu opens from the top menubar',
  { tag: '@desktop-menubar' },
  async ({ tubiHome }) => {
    skipOnMobile('Desktop menubar navigation');

    await tubiHome.goto();
    await tubiHome.browseMenuItem('Browse');

    await expect(tubiHome.browseMenuTrigger).toHaveAttribute('aria-expanded', 'true');
    await expect(tubiHome.browseCategoriesMenu).toBeVisible();
    await expect(tubiHome.browseCategoriesMenu.getByRole('menuitem').first()).toBeVisible();
  },
);

test(
  'Movies page reached from the menubar shows content tiles',
  { tag: '@desktop-menubar' },
  async ({ tubiHome, page }) => {
    skipOnMobile('Movies menubar navigation is desktop-only');
    skipInHeadless('Tubi lazy-loaded tiles need headed mode');

    await tubiHome.goto();
    await tubiHome.browseMenuItem('Movies');

    await expect(page).toHaveURL(/\/movies/);
    await tubiHome.waitForCarouselsLoaded();
    expect(await tubiHome.contentTiles.count()).toBeGreaterThanOrEqual(12);
  },
);

test(
  'Clicking a tile on the Movies page opens a detail page',
  { tag: '@desktop-menubar' },
  async ({ tubiHome, page }) => {
    skipOnMobile('Movies menubar navigation is desktop-only');
    skipInHeadless('Tubi lazy-loaded tiles need headed mode');

    await tubiHome.goto();
    await tubiHome.browseMenuItem('Movies');
    await tubiHome.waitForCarouselsLoaded();

    await tubiHome.openFirstContentTile();

    await expect(page).toHaveURL(/\/movies\/\d+/);
    await expect(tubiHome.detailTitleHeading).toBeVisible();
  },
);
